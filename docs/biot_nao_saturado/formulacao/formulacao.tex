\section*{Resumo}

O documento parte de duas fontes:
\begin{itemize}
\item \textbf{[N]} as notas \emph{Formulação poroelástica não saturada: esqueleto deformável, água e ar}
(\path{docs/biot_nao_saturado/biot_nao_saturado_agua_ar3.tex}), citadas por seção, \nota{§3};
\item \textbf{[A]} o artigo de Lira Cecílio (2026), \emph{Analytical return mapping for Mohr--Coulomb plasticity in
rotated Haigh--Westergaard space with consistent tangent operator}, J.\ Eng.\ Math.\ 157:10, citado por equação,
\art{Eq.~57}.
\end{itemize}
Ao lado de cada bloco está o trecho do NeoPZ que o resolve, no branch \code{UnsaturatedSoil}, como
\loc{arquivo}{linhas}. Os itens marcados \disc{D$k$} estão na tabela de discrepâncias (Seção~\ref{sec:disc}).

\textbf{Decisões aplicadas no código} (resposta às três decisões da versão anterior):
\begin{enumerate}
\item $\chi=S_w$ é o padrão do material, como nas notas.
\item O termo de compressibilidade do ar usa $S_a^{n}$, e não $S_a^{n+1}$. Com isso a massa de ar se conserva
exatamente no nível discreto (Seção~\ref{sec:armar}).
\item O Mohr--Coulomb é o PV2 do artigo, com $\psi=\phi$. Os projetos antigos, inclusive o executável com a classe
iterativa não associada, foram apagados.
\end{enumerate}

\textbf{Verificação.} Seis projetos de teste (Seção~\ref{sec:verif}) cobrem o ponto material, a fase água, a fase ar, o
modelo acoplado, o SRM e o GIM. Todos os testes rápidos passam. O Jacobiano completo tem ordem 2 no teste de Taylor,
inclusive com plasticidade; Terzaghi e Mandel ficam abaixo de 0,4\% de erro, e o ensaio de Liakopoulos fica tão perto
dos dados quanto a solução de Callari e Abati (2009). O SRM e o GIM do talude seco diminuem com o refino e, na malha
mais fina (52\,422 equações), ficam 0,4\% e 0,5\% acima do Bishop simplificado. Duas referências usadas nos testes têm
inconsistências, registradas na Seção~\ref{sec:verif}: o coeficiente de adensamento da solução de Mandel em Haghighat et
al.\ (2021) e o sinal da vazão na solução de Gardner em Li et al.\ (2024).

\section{Hipóteses, variáveis primárias e convenções}\label{sec:hip}
%======================================================================

As notas \nota{§1} adotam pequenas deformações, regime quase estático, temperatura constante e duas fases fluidas
imiscíveis (água e ar) sem mudança de fase. O fluxo segue Darcy generalizado com permeabilidades relativas, e a
contribuição das pressões à tensão total segue Bishop, como escolha constitutiva. As variáveis primárias são
\begin{equation}
\boxed{\mathbf u(\mathbf x,t),\qquad p_w(\mathbf x,t),\qquad p_a(\mathbf x,t),}\qquad
\mathbf z=(\mathbf u,p_w,p_a).
\end{equation}
O que o código acrescenta ou fixa:
\begin{itemize}
\item \textbf{Esqueleto elastoplástico.} As notas tratam o esqueleto como elástico \nota{§2, §4}; no código a tensão
efetiva vem do Mohr--Coulomb associado do artigo (Seção~\ref{sec:mc}).
\item \textbf{Estado plano de deformação}, com $\varepsilon_{xz}=\varepsilon_{yz}=\varepsilon_{zz}=0$; o modelo constitutivo
trabalha com o tensor 3D e devolve $\sigma_{zz}$. \loc{TPZMatUnsaturatedUPwPa.cpp}{186--200}
\item \textbf{Sinais.} Tensões positivas em tração; $p_w$ e $p_a$ manométricas e positivas em compressão. O peso entra
por pesos específicos $\gamma$ e pelo vetor unitário $\mathbf g$.
\item \textbf{Deformação em Voigt com distorção de engenharia}, na ordem $xx,xy,xz,yy,yz,zz$, como o mapa de
deformação do artigo \art{Eq.~22} e a matriz elástica \art{Eq.~26}. \loc{TPZElasticResponse.cpp}{68--87, 127--140}
\end{itemize}

\section{Sucção, saturação e pressão equivalente}\label{sec:succao}
%======================================================================

\nota{§3}:
\begin{equation}
\boxed{p_c=p_a-p_w,\qquad S_w=S_w(p_c),\qquad S_a=1-S_w,\qquad
\Gamma=\frac{\partial S_w}{\partial p_w}=-\frac{\partial S_w}{\partial p_a}=-\frac{\dd S_w}{\dd p_c}\ge0.}
\end{equation}
Para $p_c\le0$, $S_w=1$ e $\Gamma=0$. Curvas do código \loc{TPZUnsaturatedHydraulics.h}{82--108}:
\begin{align}
\text{van Genuchten (notas):}\quad & S_w=S_{wr}+(1-S_{wr})\big[1+(a_{\mathrm{VG}}p_c)^{b_{\mathrm{VG}}}\big]^{-m_{\mathrm{VG}}},
\quad m_{\mathrm{VG}}=1-1/b_{\mathrm{VG}},\\
\text{Fredlund--Xing, } C(\psi)=1:\quad & S_w=\big[\ln\big(e+(p_c/a)^{n}\big)\big]^{-m},\\
\text{Liakopoulos:}\quad & S_w=\max\big(1-c_L\,p_c^{\,b_L},\,S_{wr}\big),\\
\text{Gardner (exponencial):}\quad & S_w=S_{wr}+(1-S_{wr})\,e^{-a_Gp_c}.
\end{align}
A lei de Gardner foi acrescentada neste branch: com $k_{rw}=S_e$ ela dá $k=k_s e^{-a_Gp_c}$, que tem solução fechada para
fluxo vertical permanente (teste~2.2).

\textbf{Pressão equivalente} \nota{§1, §3}, padrão do código:
\begin{equation}
\boxed{p_B=S_wp_w+S_ap_a=p_a-S_wp_c.}
\label{eq:pB}
\end{equation}
As opções $\chi=S_e$ e $\chi$ constante continuam disponíveis. \loc{TPZMatUnsaturatedUPwPa.cpp}{97--118}
\loc{TPZMatUnsaturatedUPwPa.h}{275}

\textbf{Coeficientes de Bishop tangentes} \nota{§13.4}:
\begin{equation}
\boxed{\chi_w=\frac{\partial p_B}{\partial p_w}=S_w-\Gamma p_c,\qquad
\chi_a=\frac{\partial p_B}{\partial p_a}=S_a+\Gamma p_c,\qquad \chi_w+\chi_a=1.}
\label{eq:chit}
\end{equation}
O teste do ponto material confere $p_B$, $\chi_w$ e $\chi_a$ por diferenças centrais (Seção~\ref{sec:verif}).

\section{Equilíbrio mecânico}\label{sec:equil}
%======================================================================

\nota{§4}:
\begin{equation}
\boxed{\Div\sig+\rho_b\mathbf g=\mathbf0,\qquad \sig=\sig'-\alpha\,p_B\,\Id.}
\label{eq:equil}
\end{equation}
As notas escrevem a forma elástica incremental $\sig=\sig_0+2G\eps+\lambda\varepsilon_v\Id-\alpha(p_B-p_{B0})\Id$. No
código, $\sig'$ é a saída do Mohr--Coulomb (Seção~\ref{sec:mc}) para a deformação medida a partir do estado inicial, e
$p_B$ entra com seu valor total. As duas escritas coincidem quando o estado inicial é calculado pelo próprio modelo: o
carregamento geostático equilibra o peso com as pressões iniciais e depois os deslocamentos são zerados sem mudar $\sig'$
($\eps^p\leftarrow\eps^p-\eps$, $\eps\leftarrow\mathbf0$), de modo que $\sig_0=\sig'_0-\alpha p_{B0}\Id$.
\loc{TPZMatUnsaturatedUPwPa.cpp}{434--442}

\textbf{Peso.} As notas usam $\rho_b=(1-n)\rho_s+n(S_w\rho_w+S_a\rho_a)$. O código usa $\gamma$ constante ou, como opção,
$\gamma=(1-n)\gamma_s+nS_w\gamma_w$, sem o peso do ar e com $n$ constante. \loc{TPZMatUnsaturatedUPwPa.cpp}{120--130}
\disc{D5}

\textbf{Fatores do peso e da pressão.} O código multiplica o peso por $\lambda_g$ e $p_B$ por $\lambda_p$. No transiente
$\lambda_g=\lambda_p=1$. No carregamento geostático $\lambda_p=\lambda_g$ cresce de 0 a 1. No aumento de gravidade
$\lambda_p=\lambda_g$ onde $p_B>0$ e $\lambda_p=1$ onde $p_B\le0$: a sucção não é escalada.
\loc{TPZMatUnsaturatedUPwPa.cpp}{232--233}

\section{Lei de Darcy e permeabilidades}\label{sec:darcy}
%======================================================================

\nota{§5}:
\begin{equation}
\boxed{\mathbf q_\pi=-\frac{\kappa\,k_{r\pi}}{\mu_\pi}(\nabla p_\pi-\rho_\pi\mathbf g),\qquad \pi\in\{w,a\}.}
\end{equation}
No código, com $\kappa=k_s\mu_w/\gamma_w$ \loc{TPZMatUnsaturatedUPwPa.cpp}{313--326}:
\begin{equation}
\begin{aligned}
\mathbf q_w&=-\lambda_w\mathbf K\mathbf h_w, & \mathbf h_w&=\nabla p_w-\gamma_w\mathbf g, & \lambda_w&=\frac{k_{rw}}{\gamma_w},\\
\mathbf q_a&=-\lambda_a\mathbf K\mathbf h_a, & \mathbf h_a&=\nabla p_a-\gamma_a\mathbf g, &
\lambda_a&=\frac{\mu_w}{\mu_a}\frac{k_{ra}}{\gamma_w}.
\end{aligned}
\label{eq:darcycod}
\end{equation}
$\mathbf K$ é constante (as notas usam Kozeny--Carman) \disc{D4}; $\gamma_a$ é constante \disc{D6}.
Permeabilidades relativas \loc{TPZUnsaturatedHydraulics.h}{117--171}:
\begin{align}
\text{Mualem, água (notas):}\quad & k_{rw}=S_e^{1/2}\big\{1-[1-S_e^{1/m}]^{m}\big\}^2,\\
\text{Mualem, ar (notas):}\quad & k_{ra}=(1-S_e)^{1/2}\big[1-S_e^{1/m}\big]^{2m},\\
\text{potência; Brooks--Corey (ar):}\quad & k_{rw}=S_e^{p};\qquad k_{ra}=(1-S_e)^2\big(1-S_e^{(2+\lambda_{BC})/\lambda_{BC}}\big),
\end{align}
com limites inferiores $k_{rw}\ge10^{-12}$ e $k_{ra}\ge10^{-4}$ (derivada nula no limite). \disc{D7}

\section{Compressibilidade e massa específica dos fluidos}\label{sec:fluidos}
%======================================================================

\nota{§6}: $\rho_w=\rho_{w0}\exp[(p_w-p_{w0})/K_w]$ e, para o ar ideal isotérmico, $\rho_a\propto\pabs$, $K_a=\pabs$. O
termo de transporte normalizado se expande como \nota{§13.2}
\begin{equation}
\frac{1}{\rho_\pi}\Div(\rho_\pi\mathbf q_\pi)=\Div\mathbf q_\pi+c_\pi\,\mathbf q_\pi\cdot\nabla p_\pi,\qquad
c_w=\frac1{K_w},\qquad c_a=\frac1{\pabs}.
\label{eq:transp}
\end{equation}
No código o termo $c_w\,\mathbf q_w\cdot\nabla p_w$ não existe, como as notas admitem \nota{§13.4}, e o termo do ar existe
completo. \loc{TPZMatUnsaturatedUPwPa.cpp}{320--326, 332}

\section{Armazenamento e acoplamento entre as fases}\label{sec:armaz}
%======================================================================

\nota{§7}, com $D_w=n-\eta S_wp_c$ e $D_a=-n-\eta S_ap_c$:
\begin{equation}
\boxed{\begin{aligned}
B_w&=\alpha S_w, & B_a&=\alpha S_a,\\
C_{ww}&=\eta S_w^2+\frac{nS_w}{K_w}+D_w\Gamma, & C_{wa}&=\eta S_wS_a-D_w\Gamma,\\
C_{aw}&=\eta S_aS_w+D_a\Gamma, & C_{aa}&=\eta S_a^2+\frac{nS_a}{\pabs}-D_a\Gamma,
\end{aligned}}
\label{eq:Cnotas}
\end{equation}
que vêm de
\begin{equation}
\dot\zeta_i=S_i(\alpha\dot\varepsilon_v+\eta\dot p_B)+n\dot S_i+\frac{nS_i}{K_i}\dot p_i,\qquad i\in\{w,a\}.
\label{eq:zeta}
\end{equation}
Com $\alpha=1$ e $\eta=0$:
\begin{equation}
B_w=S_w,\quad B_a=S_a,\quad
C_{ww}=\frac{nS_w}{K_w}+n\Gamma,\quad C_{wa}=C_{aw}=-n\Gamma,\quad C_{aa}=\frac{nS_a}{\pabs}+n\Gamma.
\label{eq:Creduzido}
\end{equation}
O código não usa os coeficientes $C_{\pi\beta}$ diretamente: discretiza no tempo cada parcela de \eqref{eq:zeta}, como
descrito nas Seções~\ref{sec:armagua} e \ref{sec:armar}.

\section{Balanços de massa e sistema acoplado}\label{sec:sistema}
%======================================================================

\nota{§8, §10}:
\begin{equation}
\boxed{\begin{aligned}
\Div\sig+\rho_b\mathbf g&=\mathbf0,\\
B_w\dot\varepsilon_v+C_{ww}\dot p_w+C_{wa}\dot p_a+\Div\mathbf q_w+c_w\,\mathbf q_w\cdot\nabla p_w&=s_w,\\
B_a\dot\varepsilon_v+C_{aw}\dot p_w+C_{aa}\dot p_a+\Div\mathbf q_a+c_a\,\mathbf q_a\cdot\nabla p_a&=s_a.
\end{aligned}}
\label{eq:sistema}
\end{equation}
\textbf{Porosidade.} As notas fecham $n=n_0+(\alpha_0-n_0)\varepsilon_v+(\alpha_0-n_0)(p_B-p_{B0})/K_s$ \nota{§9}; o código
mantém $n=n_0$ nos coeficientes, e a variação do volume dos poros entra só por $B_\pi\dot\varepsilon_v$.
\loc{TPZMatUnsaturatedUPwPa.cpp}{300} \disc{D3}

\textbf{Limite saturado} \nota{§12}. Com $S_w=1$ e $\Gamma=0$ a equação da água é a de Biot saturado,
$\alpha\dot\varepsilon_v+(n/K_w)\dot p_w+\Div\mathbf q_w=0$. Os testes de Terzaghi e de Mandel verificam este limite.

\textbf{Desaparecimento do ar.} Quando $S_a\to0$, $C_{aa}$ e $\mathbf H_a$ tendem a zero \nota{§13.9}. O código adota
$k_{ra}\ge k_{ra}^{\min}$ e mantém $p_a$ como incógnita em todo o domínio.

\section{O operador \texorpdfstring{$\langle\cdot\rangle$}{<>}: condensação do armazenamento}\label{sec:cond}
%======================================================================

O símbolo $\cond{f}$ que aparece nos resíduos é o termo de armazenamento \emph{condensado nos vértices} (``lumping'').
Ele substitui a integral de Galerkin
\begin{equation}
\int_\Omega N_j\,f(p_{w,h},p_{a,h})\,\dd\Omega
\label{eq:condcons}
\end{equation}
por
\begin{equation}
\boxed{\cond{f}_j=m_j\,f(p_{w,j},p_{a,j}),\qquad m_j=\int_\Omega N_j\,\dd\Omega,}
\label{eq:conddef}
\end{equation}
em que $p_{w,j}$ e $p_{a,j}$ são as pressões do vértice $j$. Na prática, $f$ é $n\Delta S_w$ ou
$nS_a^n\Delta p_a/\pabs$.

\textbf{De onde vem.} Interpolando $f$ com as funções lineares, $f\approx\sum_kf_kN_k$, a integral
\eqref{eq:condcons} vira
\begin{equation}
\int_\Omega N_jf\,\dd\Omega\approx\sum_kM_{jk}f_k,\qquad M_{jk}=\int_\Omega N_jN_k\,\dd\Omega .
\label{eq:condM}
\end{equation}
A condensação por soma de linhas troca $M_{jk}$ por $\delta_{jk}\sum_lM_{jl}$. Como $\sum_lN_l=1$,
\begin{equation}
\sum_lM_{jl}=\int_\Omega N_j\,\dd\Omega=m_j\quad\Longrightarrow\quad \sum_kM_{jk}f_k\ \to\ m_jf_j=\cond{f}_j .
\label{eq:condrow}
\end{equation}
O mesmo resultado sai da quadratura nos vértices, $\int_eg\,\dd\Omega\approx\tfrac{A_e}{3}\sum_vg(\mathbf x_v)$, aplicada a
$g=N_jf$, porque $N_j(\mathbf x_v)=\delta_{jv}$. As notas propõem a forma
$\operatorname{diag}(\int C_{\pi\beta}N_i\,\dd\Omega)$ \nota{§14}, que coincide com \eqref{eq:conddef} quando o coeficiente é
constante; o código avalia a função no vértice \disc{D13}.

\textbf{Por que condensar.} As notas registram que a diagonalização elimina oscilações não físicas em frentes de
saturação e preserva a positividade de $S_w$ \nota{§13.9}. O capítulo~2 da dissertação da PUC-Rio enviada (arquivo
\path{12189_3}, p.~16) atribui a Celia et al.\ (1990) a conclusão de que a diagonalização é ``condição necessária e
suficiente'' para eliminar as oscilações da carga de pressão. Há um terceiro motivo, próprio deste código: com
\eqref{eq:conddef} a soma das equações nodais dá exatamente a variação de um volume armazenado definido pelos valores
nodais, o que torna exata a conservação discreta da água e do ar (Seções~\ref{sec:armagua} e \ref{sec:armar}).

\textbf{No código.} \code{VertexValues} reconstrói as pressões dos três vértices a partir do valor e do gradiente no ponto
de integração, o que é exato para o P1. O termo do vértice $j$ é somado nos três pontos com peso $N_j(\mathbf x_g)w_g$, e
$\sum_gN_j(\mathbf x_g)w_g=m_j^e$ exatamente.
\loc{TPZMatUnsaturatedUPwPa.cpp}{132--151, 235--240, 338--354} A derivada é diagonal:
\begin{equation}
\frac{\partial\cond{f}_j}{\partial p_k}=\delta_{jk}\,m_j\,\frac{\partial f}{\partial p}(p_j).
\label{eq:condder}
\end{equation}
A condensação só é usada com pressões P1; sem ela, o armazenamento é integrado nos pontos.

\section{Armazenamento da água}\label{sec:armagua}
%======================================================================

A massa de água por volume da mistura é
\begin{equation}
m_w=\rho_w\,n\,S_w .
\label{eq:mw}
\end{equation}
Dividindo o balanço de massa por $\rho_w$, a taxa de armazenamento é a Eq.~\eqref{eq:zeta} das notas com $i=w$:
\begin{equation}
\dot\zeta_w=\underbrace{S_w\alpha\dot\varepsilon_v}_{\text{(a) poros}}+\underbrace{S_w\eta\dot p_B}_{\text{(b) grãos}}
+\underbrace{n\dot S_w}_{\text{(c) saturação}}+\underbrace{\frac{nS_w}{K_w}\dot p_w}_{\text{(d) água}} .
\label{eq:zetaw}
\end{equation}
(a) é a variação do volume dos poros ocupada pela fração $S_w$; (b), a compressão dos grãos (nula com $\eta=0$); (c), a
troca de ar por água nos poros; (d), a compressão da água.

\textbf{Parcela (c): forma em taxa e forma secante.} Pela regra da cadeia, $n\dot S_w=n\Gamma(\dot p_w-\dot p_a)$, que é a
origem de $C_{ww}$ e $C_{wa}$ em \eqref{eq:Creduzido}. Com Euler implícito há duas escritas:
\begin{align}
\text{em taxa (notas \nota{§13.7}):}\quad & n\,\Gamma^{n+1}(\Delta p_w-\Delta p_a),\label{eq:wtaxa}\\
\text{secante (código):}\quad & n\,(S_w^{n+1}-S_w^n).\label{eq:wsec}
\end{align}
Como $\Delta S_w=S_w(p_c^{n+1})-S_w(p_c^n)=-\Gamma^{n+1}\Delta p_c+O(\Delta p_c^2)$, as duas diferem em
$O(\Delta p_c^2)$ por passo. A forma secante é a forma mista de Celia et al.\ (1990), referência [16] de Callari e Abati
(2009, p.~17); Callari e Abati (2009, p.~5) registram que a forma em taxa ``can lead to unacceptable errors in fluid
mass balance, especially in the case of strongly non-linear response, typically characterizing infiltration
problems''. \disc{D8}

\textbf{Conservação discreta.} Com esqueleto rígido ($\Theta=0$) e $K_w\to\infty$, a equação da água no vértice $j$ é
$\cond{n\Delta S_w}_j+\Delta t\int\nabla N_j\cdot\lambda_w\mathbf K\mathbf h_w\,\dd\Omega-\Delta t\,Q_j=0$, com $Q_j$ a vazão
nodal que entra. Somando em $j$ e usando $\sum_j\nabla N_j=\mathbf0$:
\begin{equation}
\sum_jm_j\,n\,(S_{w,j}^{n+1}-S_{w,j}^n)=\Delta t\sum_jQ_j .
\label{eq:wcons}
\end{equation}
O volume armazenado $V_w=\sum_jm_jnS_{w,j}$ muda exatamente pela vazão que entra, até a tolerância do Newton. No teste
2.3 o erro relativo foi $2\times10^{-14}$. Com a forma em taxa \eqref{eq:wtaxa} a soma daria
$\sum_jm_jn\Gamma_j^{n+1}\Delta p_{c,j}$, que não é a variação de nenhum volume.

\textbf{Demais parcelas no código.} (a) e (b) entram como $S_w^{n+1}\Theta$, com $\Theta=\alpha\Delta\varepsilon_v+\eta\Delta p_B$,
nos pontos de integração; (d) entra como $nS_w^{n+1}\Delta p_w/K_w$, também nos pontos. \loc{TPZMatUnsaturatedUPwPa.cpp}{301--312, 331, 343}

\section{Armazenamento do ar}\label{sec:armar}
%======================================================================

A massa de ar por volume da mistura e a lei do gás ideal isotérmico dão
\begin{equation}
m_a=\rho_a\,n\,S_a,\qquad \rho_a=\rho_{a0}\,\frac{\pabs}{\pabs_0},\qquad \pabs=p_a+p_{\mathrm{atm}} .
\label{eq:ma}
\end{equation}
O balanço dividido por $\rho_a$ é a Eq.~\eqref{eq:zeta} com $i=a$, mais o termo $c_a\mathbf q_a\cdot\nabla p_a$ de
\eqref{eq:transp}. No tempo, a variação de massa no passo dividida por $\rho_a^{n+1}$ é
\begin{equation}
\frac{m_a^{n+1}-m_a^n}{\rho_a^{n+1}}=n\Big(S_a^{n+1}-S_a^n\frac{\pabs{}^{n}}{\pabs{}^{n+1}}\Big)
=\boxed{n\,\Delta S_a+n\,S_a^{n}\,\frac{\Delta p_a}{\pabs{}^{n+1}}} .
\label{eq:asec}
\end{equation}
É a forma usada agora pelo código, condensada nos vértices: $\cond{-n\Delta S_w+nS_a^n\Delta p_a/\pabs}$.
\loc{TPZMatUnsaturatedUPwPa.cpp}{336--337, 345} A versão anterior usava $S_a^{n+1}$ no segundo termo, o que acrescenta
\begin{equation}
n\,\Delta S_a\,\frac{\Delta p_a}{\pabs{}^{n+1}}
\label{eq:aerro}
\end{equation}
por passo. Nas rodadas do manuscrito esse termo deu erro de 0,25--0,34\% no balanço global de ar.

\textbf{Conservação discreta exata.} Multiplica-se a equação do ar de cada vértice por $\pabs_j{}^{n+1}$ e soma-se, com
esqueleto rígido. O armazenamento dá
\begin{equation}
\sum_j\pabs_j{}^{n+1}m_jn\Big(\Delta S_{a,j}+S_{a,j}^n\frac{\Delta p_{a,j}}{\pabs_j{}^{n+1}}\Big)
=\sum_jm_jn\big[(S_a\pabs)_j^{n+1}-(S_a\pabs)_j^{n}\big]=\Delta M,\qquad M=\sum_jm_jnS_{a,j}\pabs_j .
\label{eq:aconsS}
\end{equation}
O transporte dá zero no interior. Com $\pabs_h=\sum_j\pabs_jN_j$, o código calcula $c_a=1/\pabs_h$ no ponto de
integração, logo $\pabs_hc_a=1$ e $\nabla\pabs_h=\nabla p_{a,h}$:
\begin{equation}
\sum_j\pabs_j\Big[\int N_jc_a\,\mathbf q_a\cdot\nabla p_a\,\dd\Omega-\int\nabla N_j\cdot\mathbf q_a\,\dd\Omega\Big]
=\int\big(\pabs_hc_a\big)\mathbf q_a\cdot\nabla p_a\,\dd\Omega-\int\nabla\pabs_h\cdot\mathbf q_a\,\dd\Omega=0 .
\label{eq:aconsT}
\end{equation}
Portanto $M$ muda só pelo ar que atravessa o contorno. Com $S_a^{n+1}$ apareceria a parcela \eqref{eq:aerro} em
\eqref{eq:aconsS}. O teste 3a (ar confinado, lei de Boyle) mantém $M$ constante com erro $8\times10^{-16}$ enquanto o
volume de ar cai a 70\% e $p_a$ sobe a 43~kPa; o teste 3c fecha o balanço com erro $5\times10^{-16}$ de $M_0$ com 35\% do
ar expulso pelo topo.

\textbf{Esqueleto deformável.} A parcela $S_a\Theta$ fica nos pontos de integração; com o peso $\pabs_j$ ela dá
$\int\pabs_hS_a\Theta\,\dd\Omega$, a variação de massa pela mudança do volume dos poros, consistente em primeira ordem.

\section{Resíduos discretos passo a passo}\label{sec:residuos}
%======================================================================

Esta seção vai das equações de campo \eqref{eq:equil} e \eqref{eq:sistema} até as linhas do código que montam o
resíduo. As notas fazem o mesmo caminho \nota{§13.2--13.7}, com as equações de massa divididas por $\Delta t$; o código
as multiplica por $\Delta t$, o que só muda a escala dessas linhas.

\textbf{Passo 1: forma fraca.} São as Eqs.~\eqref{eq:fracau} e \eqref{eq:fracap}, com $\rho_b\mathbf g$ escrito como
$\gamma\mathbf g$ ($\gamma$ em kN/m$^3$, $\mathbf g$ unitário).

\textbf{Passo 2: Galerkin.} As funções-teste são as próprias funções de forma: $\mathbf v=\mathbf N_u\,\delta\mathbf U$ e
$w=N_j$, um vértice $j$ de cada vez. As incógnitas são interpoladas por
\begin{equation}
\mathbf u_h=\mathbf N_u\mathbf U,\qquad p_{\pi,h}=\mathbf N_p\mathbf P_\pi,\qquad
\eps_h=\mathbf B_u\mathbf U,\qquad \varepsilon_{v,h}=\mathbf m^{\mathsf T}\mathbf B_u\mathbf U,\qquad
\nabla p_{\pi,h}=\mathbf B_p\mathbf P_\pi .
\label{eq:galerkin}
\end{equation}
Com a deformação em Voigt e a distorção de engenharia, $\eps(\mathbf v):\sig=\delta\mathbf U^{\mathsf T}\mathbf B_u^{\mathsf T}\sig$.
Como $\delta\mathbf U$ é arbitrário, \eqref{eq:fracau} dá uma equação por grau de liberdade de deslocamento:
\begin{equation}
\int_\Omega\mathbf B_u^{\mathsf T}\sig\,\dd\Omega-\int_\Omega\mathbf N_u^{\mathsf T}\gamma\,\mathbf g\,\dd\Omega
-\int_{\Gamma_t}\mathbf N_u^{\mathsf T}\bar{\mathbf t}\,\dd\Gamma=\mathbf 0 .
\label{eq:galu}
\end{equation}

\textbf{Passo 3: tensão efetiva.} Com $\sig=\sig'-\alpha p_B\Id$ \eqref{eq:equil} e $\Id\to\mathbf m$ em Voigt,
\begin{equation}
\mathbf B_u^{\mathsf T}\sig=\mathbf B_u^{\mathsf T}\sig'-\alpha\,p_B\,\mathbf B_u^{\mathsf T}\mathbf m .
\label{eq:Bsig}
\end{equation}
Os fatores $\lambda_p$ e $\lambda_g$ da Seção~\ref{sec:equil} multiplicam $p_B$ e o peso.

\textbf{Passo 4: Euler implícito nas equações de massa.} Integra-se \eqref{eq:fracap} de $t_n$ a $t_{n+1}$. O
armazenamento vira a variação no passo e os fluxos ficam em $t_{n+1}$:
\begin{equation}
\int_{t_n}^{t_{n+1}}\!\!\int_\Omega N_j\,\dot\zeta_\pi\,\dd\Omega\,\dd t=\int_\Omega N_j\,\Delta\zeta_\pi\,\dd\Omega,\qquad
\int_{t_n}^{t_{n+1}}(\cdots)\,\dd t\approx\Delta t\,(\cdots)^{n+1}\ \text{nos demais termos}.
\label{eq:euler}
\end{equation}
Resulta, para cada vértice $j$ e cada fase,
\begin{equation}
\int_\Omega N_j\,\Delta\zeta_\pi\,\dd\Omega-\Delta t\int_\Omega\nabla N_j\cdot\mathbf q_\pi\,\dd\Omega
+\Delta t\int_\Omega N_j\,c_\pi\,\mathbf q_\pi\cdot\nabla p_\pi\,\dd\Omega-\Delta t\,f_{\pi,j}=0,
\label{eq:galp}
\end{equation}
em que $f_{\pi,j}$ é a vazão que entra pelo contorno associada ao vértice $j$ (convenção do código, \disc{D20}). Como em
\eqref{eq:fracap}, é o balanço de massa dividido por $\rho_\pi$: cada termo é um volume por unidade de espessura.

\textbf{Passo 5: as parcelas de $\Delta\zeta$.} Cada parcela de \eqref{eq:zeta} é discretizada no tempo como nas Seções
\ref{sec:armagua} e \ref{sec:armar}:
\begin{equation}
\begin{aligned}
\Delta\zeta_w&=S_w^{n+1}\,\Theta+n\,\Delta S_w+\frac{n\,S_w^{n+1}}{K_w}\,\Delta p_w,\\
\Delta\zeta_a&=S_a^{n+1}\,\Theta-n\,\Delta S_w+\frac{n\,S_a^{n}}{\pabs{}^{n+1}}\,\Delta p_a,
\end{aligned}
\qquad \Theta=\alpha\,\Delta\varepsilon_v+\eta\,\Delta p_B .
\label{eq:dzeta}
\end{equation}
$\Theta$ é a variável \code{X} do código. \loc{TPZMatUnsaturatedUPwPa.cpp}{301--312}

\textbf{Passo 6: condensação.} A parcela de saturação e a de compressibilidade do ar vão para os vértices com o operador
da Seção~\ref{sec:cond}; as outras ficam nos pontos de integração:
\begin{equation}
\int_\Omega N_j\,\Delta\zeta_w\,\dd\Omega\ \to\ \int_\Omega N_j\Big(S_w\Theta+\frac{nS_w}{K_w}\Delta p_w\Big)\dd\Omega
+\cond{n\,\Delta S_w}_j,
\label{eq:lumpw}
\end{equation}
\begin{equation}
\int_\Omega N_j\,\Delta\zeta_a\,\dd\Omega\ \to\ \int_\Omega N_j\,S_a\Theta\,\dd\Omega
+\cond{-n\,\Delta S_w+\frac{n\,S_a^n}{\pabs}\Delta p_a}_j .
\label{eq:lumpa}
\end{equation}
Sem condensação (\code{SetLumpedStorage(false)}) os termos entre $\langle\cdot\rangle$ são integrados nos pontos, com
$S_w^n$ calculado no ponto a partir das pressões convergidas. \loc{TPZMatUnsaturatedUPwPa.cpp}{328, 333--337}

\textbf{Passo 7: Darcy.} Com \eqref{eq:darcycod}, $-\Delta t\int\nabla N_j\cdot\mathbf q_\pi\,\dd\Omega=
\Delta t\int\nabla N_j\cdot\lambda_\pi\mathbf K\mathbf h_\pi\,\dd\Omega$. O termo do ar de \eqref{eq:transp} fica
\begin{equation}
\Delta t\int_\Omega N_j\,c_a\,\mathbf q_a\cdot\nabla p_a\,\dd\Omega=-\Delta t\int_\Omega N_j\,c_a\lambda_a\,
\mathrm{adv}\,\dd\Omega,\qquad \mathrm{adv}=(\mathbf K\mathbf h_a)\cdot\nabla p_a ;
\label{eq:advterm}
\end{equation}
o termo da água não existe no código \disc{D10}. \loc{TPZMatUnsaturatedUPwPa.cpp}{313--325}

\textbf{Passo 8: resíduos.} Juntando os passos, com $\mathsf A_e$ a montagem dos elementos e as grandezas sem índice de
tempo em $t_{n+1}$:
\begin{equation}
\boxed{\mathbf R_u=\mathsf A_e\Big[\int_{\Omega^e}\mathbf B_u^{\mathsf T}\sig'\,\dd\Omega
-\lambda_p\,\alpha\int_{\Omega^e}\mathbf B_u^{\mathsf T}\mathbf m\,p_B\,\dd\Omega
-\lambda_g\int_{\Omega^e}\mathbf N_u^{\mathsf T}\gamma\,\mathbf g\,\dd\Omega\Big]-\mathbf f_t,}
\label{eq:Ru}
\end{equation}
\begin{equation}
\boxed{\mathbf R_w=\mathsf A_e\Big[\int_{\Omega^e}\mathbf N_p^{\mathsf T}\Big(S_w\Theta+\frac{nS_w}{K_w}\Delta p_w\Big)\dd\Omega
+\Delta t\int_{\Omega^e}\mathbf B_p^{\mathsf T}\lambda_w\mathbf K\mathbf h_w\,\dd\Omega\Big]
+\cond{n\,\Delta S_w}-\Delta t\,\mathbf f_w,}
\label{eq:Rw}
\end{equation}
\begin{equation}
\boxed{\begin{aligned}\mathbf R_a={}&\mathsf A_e\Big[\int_{\Omega^e}\mathbf N_p^{\mathsf T}\big(S_a\Theta-\Delta t\,c_a\lambda_a\,\mathrm{adv}\big)\dd\Omega
+\Delta t\int_{\Omega^e}\mathbf B_p^{\mathsf T}\lambda_a\mathbf K\mathbf h_a\,\dd\Omega\Big]\\
&+\cond{-n\,\Delta S_w+\frac{nS_a^n}{\pabs}\Delta p_a}-\Delta t\,\mathbf f_a .\end{aligned}}
\label{eq:Ra}
\end{equation}
$\mathbf f_t=\int_{\Gamma_t}\mathbf N_u^{\mathsf T}\bar{\mathbf t}\,\dd\Gamma$, e $\mathbf f_\pi$ reúne as vazões prescritas que
entram. $\cond{\cdot}$ é o vetor dos termos condensados \eqref{eq:conddef}, um por vértice.

\textbf{Passo 9: o que o código monta.} O NeoPZ monta $-\mathbf R$ (vetor \code{ef}) e $\partial\mathbf R/\partial\mathbf X$
(matriz \code{ek}); em cada ponto de integração as parcelas são multiplicadas pelo peso \code{dv}. Correspondência
(arquivo \path{TPZMatUnsaturatedUPwPa.cpp}, salvo indicação):

\section{Jacobiano passo a passo}\label{sec:jac}
%======================================================================

O Newton precisa de $\mathbf J=\partial\mathbf R/\partial\mathbf X$, com $\mathbf X=(\mathbf U,\mathbf P_w,\mathbf P_a)$. O
código deriva \eqref{eq:Ru}--\eqref{eq:Ra} por inteiro (Newton completo). As notas admitem omitir as derivadas dos
coeficientes multiplicadas pelos incrementos do passo (Newton modificado) \nota{§13.7} \disc{D11}. As grandezas do passo
anterior ($p^n$, $S_w^n$, $\varepsilon_v^n$, estado plástico convergido) são constantes na derivação.

\textbf{Passo 1: derivadas elementares no ponto de integração.}
\begin{equation}
\frac{\partial\eps}{\partial\mathbf U}=\mathbf B_u,\qquad
\frac{\partial\varepsilon_v}{\partial\mathbf U}=\mathbf m^{\mathsf T}\mathbf B_u,\qquad
\frac{\partial p_\pi}{\partial\mathbf P_\pi}=\mathbf N_p,\qquad
\frac{\partial\nabla p_\pi}{\partial\mathbf P_\pi}=\frac{\partial\mathbf h_\pi}{\partial\mathbf P_\pi}=\mathbf B_p,
\label{eq:d1}
\end{equation}
\begin{equation}
\frac{\partial\sig'}{\partial\eps}=\mathbb D\quad\text{(Seção~\ref{sec:mctan}; }\mathbb D=\mathbb C\text{ nos pontos elásticos)},
\label{eq:d2}
\end{equation}
\begin{equation}
\frac{\partial S_w}{\partial p_w}=\Gamma,\qquad \frac{\partial S_w}{\partial p_a}=-\Gamma,\qquad
\frac{\partial S_a}{\partial p_\pi}=-\frac{\partial S_w}{\partial p_\pi},\qquad
\frac{\partial p_B}{\partial p_\pi}=\chi_\pi\ \ \eqref{eq:chit},
\label{eq:d3}
\end{equation}
\begin{equation}
\frac{\partial\Theta}{\partial\mathbf U}=\alpha\,\mathbf m^{\mathsf T}\mathbf B_u,\qquad
\frac{\partial\Theta}{\partial p_\pi}=\eta\,\chi_\pi,
\label{eq:d4}
\end{equation}
\begin{equation}
\frac{\partial\gamma}{\partial p_w}=-\frac{\partial\gamma}{\partial p_a}=n\,\gamma_w\,\Gamma\quad
\text{(peso da mistura; zero com }\gamma\text{ constante)},
\label{eq:d5}
\end{equation}
\begin{equation}
\lambda'_\pi=\frac{\partial\lambda_\pi}{\partial p_w}=-\frac{\partial\lambda_\pi}{\partial p_a}
=\lambda_{\pi0}\,\frac{\dd k_{r\pi}}{\dd S_w}\,\Gamma,\qquad \lambda_{w0}=\frac1{\gamma_w},\quad
\lambda_{a0}=\frac{\mu_w}{\mu_a}\,\frac1{\gamma_w},
\label{eq:d6}
\end{equation}
\begin{equation}
\frac{\partial c_a}{\partial p_a}=-c_a^2,\qquad
\frac{\partial\,\mathrm{adv}}{\partial\mathbf P_a}=\big[\mathbf K(\mathbf h_a+\nabla p_a)\big]^{\mathsf T}\mathbf B_p
\quad(\mathbf K\text{ simétrico}).
\label{eq:d7}
\end{equation}
$\lambda_p$ e $\lambda_g$ não dependem de $\mathbf X$. Com a sucção não escalada, $\lambda_p$ salta só onde $p_B=0$, e ali
$\lambda_pp_B$ é contínuo. A derivada do termo condensado é diagonal \eqref{eq:condder}.
\loc{TPZMatUnsaturatedUPwPa.cpp}{226--233, 313--325}

\textbf{Passo 2: linha do equilíbrio.} Derivando \eqref{eq:Ru} com \eqref{eq:d1}--\eqref{eq:d5}:
\begin{equation}
\boxed{\mathbf J_{uu}=\int\mathbf B_u^{\mathsf T}\,\mathbb D\,\mathbf B_u\,\dd\Omega,}
\label{eq:Juu}
\end{equation}
\begin{equation}
\boxed{\mathbf J_{u\pi}=-\int\Big[\lambda_p\,\alpha\,\chi_\pi\,\mathbf B_u^{\mathsf T}\mathbf m
+\lambda_g\,\frac{\partial\gamma}{\partial p_\pi}\,\mathbf N_u^{\mathsf T}\mathbf g\Big]\mathbf N_p\,\dd\Omega,
\qquad\pi\in\{w,a\}.}
\label{eq:Jup}
\end{equation}

\loc{TPZMatUnsaturatedUPwPa.cpp}{297}, e as equações de pressão são eliminadas pelo filtro.

\textbf{Passo 3: linhas de massa, colunas de $\mathbf U$.} Só $\Theta$ depende de $\mathbf U$:
\begin{equation}

\mathbf J_{au}=\int\mathbf N_p^{\mathsf T}S_a\,\alpha\,\mathbf m^{\mathsf T}\mathbf B_u\,\dd\Omega .}

\mathbf J_{wa}={}&\int\mathbf N_p^{\mathsf T}\Big[-\Gamma\Theta+S_w\eta\chi_a-\frac{n}{K_w}\Gamma\Delta p_w\Big]\mathbf N_p\,\dd\Omega\\
&-\Delta t\int\mathbf B_p^{\mathsf T}(\mathbf K\mathbf h_w)\,\lambda'_w\,\mathbf N_p\,\dd\Omega-\operatorname{diag}\cond{n\Gamma},
\end{aligned}}
\label{eq:Jwa}
\end{equation}
\begin{equation}
\boxed{\begin{aligned}
\mathbf J_{aw}={}&\int\mathbf N_p^{\mathsf T}\big[-\Gamma\Theta+S_a\eta\chi_w-\Delta t\,c_a\lambda'_a\,\mathrm{adv}\big]\mathbf N_p\,\dd\Omega\\
&+\Delta t\int\mathbf B_p^{\mathsf T}(\mathbf K\mathbf h_a)\,\lambda'_a\,\mathbf N_p\,\dd\Omega-\operatorname{diag}\cond{n\Gamma},
\end{aligned}}
\label{eq:Jaw}
\end{equation}
\begin{equation}
\boxed{\begin{aligned}
\mathbf J_{aa}={}&\int\mathbf N_p^{\mathsf T}\big[\Gamma\Theta+S_a\eta\chi_a+\Delta t\,(c_a^2\lambda_a+c_a\lambda'_a)\,\mathrm{adv}\big]\mathbf N_p\,\dd\Omega\\
&-\Delta t\int\mathbf N_p^{\mathsf T}c_a\lambda_a\big[\mathbf K(\mathbf h_a+\nabla p_a)\big]^{\mathsf T}\mathbf B_p\,\dd\Omega
+\Delta t\int\mathbf B_p^{\mathsf T}\lambda_a\mathbf K\mathbf B_p\,\dd\Omega\\
&-\Delta t\int\mathbf B_p^{\mathsf T}(\mathbf K\mathbf h_a)\,\lambda'_a\,\mathbf N_p\,\dd\Omega
+\operatorname{diag}\cond{n\Gamma+\frac{nS_a^n}{\pabs}-\frac{nS_a^n\,\Delta p_a}{\pabs^2}} .
\end{aligned}}
\label{eq:Jaa}
\end{equation}
\loc{TPZMatUnsaturatedUPwPa.cpp}{346--352, 362--381} Sem condensação, os termos $\operatorname{diag}\cond{\cdot}$ viram
integrais nos pontos, $\int\mathbf N_p^{\mathsf T}(\cdot)\mathbf N_p\,\dd\Omega$, com $S_a^n$ e $c_a$ do ponto
\loc{TPZMatUnsaturatedUPwPa.cpp}{372--376}.

\textbf{Passo 5: confronto com as notas.} Sem as parcelas proporcionais aos incrementos ($\Gamma\Theta$,
$\Gamma\Delta p_w$, $\Delta p_a/\pabs^2$), os coeficientes de $\mathbf N_p^{\mathsf T}(\cdot)\mathbf N_p$ somados aos termos
condensados são os coeficientes $C_{\pi\beta}$ das notas \eqref{eq:Cnotas}. Com $\chi_w=S_w-\Gamma p_c$,
$\chi_a=S_a+\Gamma p_c$, $D_w=n-\eta S_wp_c$ e $D_a=-n-\eta S_ap_c$:
\begin{equation}
\begin{aligned}
S_w\eta\chi_w+\frac{nS_w}{K_w}+n\Gamma&=\eta S_w^2+\frac{nS_w}{K_w}+D_w\Gamma=C_{ww},\\
S_w\eta\chi_a-n\Gamma&=\eta S_wS_a-D_w\Gamma=C_{wa},\\
S_a\eta\chi_w-n\Gamma&=\eta S_aS_w+D_a\Gamma=C_{aw},\\
S_a\eta\chi_a+\frac{nS_a}{\pabs}+n\Gamma&=\eta S_a^2+\frac{nS_a}{\pabs}-D_a\Gamma=C_{aa}.
\end{aligned}
\label{eq:Cident}
\end{equation}
No código, $C_{aa}$ usa $S_a^n$ (Seção~\ref{sec:armar}). Os demais blocos coincidem com os das notas \nota{§13.4, §13.7}:
$\mathbf J_{uu}=\mathbf K$ (sem $\mathbf G_{uu}$, porque $n$ é constante, \disc{D3}), $\mathbf J_{u\pi}=-\mathbf Q_{u\pi}-\mathbf G_{u\pi}$,
$\mathbf J_{\pi u}=\mathbf Q_{\pi u}$, e os termos com $\lambda'_\pi$ e com $c_a$ são os blocos $\tilde{\mathbf H}_{\pi\beta}$ (sem a parcela com
$\dd\rho_\pi/\dd p_\pi$, porque $\gamma_w$ e $\gamma_a$ são constantes, \disc{D6}), tudo multiplicado por $\Delta t$ nas
linhas de massa.

\textbf{Passo 6: verificação.} O teste de Taylor compara o resíduo perturbado com a previsão linear,
\begin{equation}
E(\epsilon)=\big\|\mathbf R(\mathbf X+\epsilon\mathbf d)-\mathbf R(\mathbf X)-\epsilon\,\mathbf J\mathbf d\big\|=O(\epsilon^2),\qquad
p=\log_2\frac{E(\epsilon)}{E(\epsilon/2)}\ \to\ 2,
\label{eq:taylor}
\end{equation}
para direções $\mathbf d$ aleatórias. \loc{Projects/UnsatCommon/UnsatTaylor.h}{18--55} Ordem 2 só sai se todos os termos de
\eqref{eq:Juu}--\eqref{eq:Jaa} estiverem certos. Resultados (Seção~\ref{sec:verif}): fase água 2,000 com e sem
condensação; água e ar 2,000; modelo completo com 96 pontos plásticos 2,000 em quatro variantes ($\chi=S_w$; $\chi$
constante; $\chi=S_w$ com peso da mistura, $\eta>0$ e armazenamento sem condensação; saturado).

\section{Por que o Jacobiano não é simétrico}\label{sec:simetria}
%======================================================================

Com os blocos das Eqs.~\eqref{eq:Juu}--\eqref{eq:Jaa},
\begin{equation}
\mathbf J=\begin{bmatrix}\mathbf J_{uu}&\mathbf J_{uw}&\mathbf J_{ua}\\ \mathbf J_{wu}&\mathbf J_{ww}&\mathbf J_{wa}\\
\mathbf J_{au}&\mathbf J_{aw}&\mathbf J_{aa}\end{bmatrix}.
\label{eq:Jblocos}
\end{equation}
A simetria exigiria $\mathbf J_{uu}=\mathbf J_{uu}^{\mathsf T}$, $\mathbf J_{ww}=\mathbf J_{ww}^{\mathsf T}$,
$\mathbf J_{aa}=\mathbf J_{aa}^{\mathsf T}$, $\mathbf J_{wa}=\mathbf J_{aw}^{\mathsf T}$ e $\mathbf J_{\pi u}=\mathbf J_{u\pi}^{\mathsf T}$.
As notas já registram que o acoplamento é em geral não simétrico porque $B_\pi=\alpha S_\pi\neq\alpha\chi_\pi$ quando
$\Gamma p_c\neq0$ \nota{§13.4}. Os blocos são examinados abaixo um a um.

\textbf{(1) Partes simétricas.}
\begin{itemize}
\item $\mathbf J_{uu}=\int\mathbf B_u^{\mathsf T}\mathbb D\mathbf B_u$ é simétrico porque o PV2 é associado e sua tangente
consistente é simétrica: $|\mathbb D-\mathbb D^{\mathsf T}|/|\mathbb D|\le6\times10^{-15}$ (Seção~\ref{sec:mcverif}).
\item Os termos de Darcy $\Delta t\int\mathbf B_p^{\mathsf T}\lambda_\pi\mathbf K\mathbf B_p$ são simétricos porque $\mathbf K$ é
simétrico.
\item O bloco condensado de cada vértice é
\begin{equation}
m_j\begin{bmatrix}n\Gamma&-n\Gamma\\-n\Gamma&n\Gamma+nS_a^n/\pabs-nS_a^n\Delta p_a/\pabs^2\end{bmatrix},
\label{eq:Jcond}
\end{equation}
simétrico. Na versão anterior, com $S_a^{n+1}$, a derivada de $nS_a^{n+1}\Delta p_a/\pabs$ em relação a $p_w$ somava
$-n\Gamma\Delta p_a/\pabs$ só à linha do ar, e este bloco não era simétrico.
\end{itemize}

\textbf{(2) Sinal das linhas de massa.} No limite saturado e linear ($S_w=1$, $\Gamma=0$, $\chi_w=1$, $\lambda_p=1$,
$\gamma$ constante, $\eta=0$; só $\mathbf u$ e $p_w$), as Eqs.~\eqref{eq:Jup} e \eqref{eq:Jpu} dão
\begin{equation}
\mathbf J_{uw}=-\mathbf Q,\qquad \mathbf J_{wu}=\mathbf Q^{\mathsf T},\qquad
\mathbf Q=\int\alpha\,\mathbf B_u^{\mathsf T}\mathbf m\,\mathbf N_p\,\dd\Omega .
\label{eq:Qsat}
\end{equation}
Logo $\mathbf J_{wu}=-\mathbf J_{uw}^{\mathsf T}$: mesmo no caso mais simples, $\mathbf J$ não é simétrico com a convenção
de sinais do código, que é a das notas \nota{§13.5, §13.7} ($-\mathbf Q_{uw}$ na linha de $\mathbf u$, $+\mathbf Q_{wu}$ na linha
de $p_w$). A simetria a que as notas se referem no limite saturado é $\mathbf Q_{wu}=\mathbf Q_{uw}^{\mathsf T}$. Para a matriz
ficar simétrica é preciso trocar o sinal das linhas de massa, o que dá a forma simétrica indefinida
\begin{equation}
\begin{bmatrix}\mathbf J_{uu}&-\mathbf Q\\-\mathbf Q^{\mathsf T}&-(\mathbf C+\Delta t\,\mathbf H)\end{bmatrix},\qquad
\mathbf C=\int\mathbf N_p^{\mathsf T}\frac{n}{K_w}\mathbf N_p\,\dd\Omega,\quad
\mathbf H=\int\mathbf B_p^{\mathsf T}\lambda_w\mathbf K\mathbf B_p\,\dd\Omega .
\label{eq:Jsimsat}
\end{equation}
O código não faz essa troca. Os itens (3) a (5) valem mesmo com ela.

\textbf{(3) Acoplamento: Bishop e peso.} Com a troca de sinal, a simetria pede $\mathbf J_{u\pi}=-\mathbf J_{\pi u}^{\mathsf T}$.
Pelas Eqs.~\eqref{eq:Jup} e \eqref{eq:Jpu},
\begin{equation}
\mathbf J_{u\pi}+\mathbf J_{\pi u}^{\mathsf T}=\int\alpha\,(S_\pi-\lambda_p\chi_\pi)\,\mathbf B_u^{\mathsf T}\mathbf m\,\mathbf N_p\,\dd\Omega
-\lambda_g\int\frac{\partial\gamma}{\partial p_\pi}\,\mathbf N_u^{\mathsf T}\mathbf g\,\mathbf N_p\,\dd\Omega .
\label{eq:assimup}
\end{equation}
Com $\chi=S_w$ e $\lambda_p=1$, $S_w-\chi_w=\Gamma p_c$ e $S_a-\chi_a=-\Gamma p_c$: a primeira integral só se anula onde
$\Gamma p_c=0$. Com $\chi$ constante, $S_w-\chi$ não se anula em geral. A segunda integral (peso da mistura) não tem
correspondente nas linhas de massa, que não dependem de $\mathbf U$ por meio de $\gamma$.

\textbf{(4) Mobilidade e massa específica do ar.} Os termos com $\lambda'_\pi$ têm o gradiente na função-teste e o valor
na função-tentativa ($\mathbf B_p^{\mathsf T}(\cdot)\mathbf N_p$); o termo de $c_a$ tem o contrário
($\mathbf N_p^{\mathsf T}(\cdot)\mathbf B_p$). São operadores do tipo advectivo e não têm transposto no Jacobiano:
\begin{equation}
\mathbf J_{ww}-\mathbf J_{ww}^{\mathsf T}=\Delta t\int\Big[\mathbf B_p^{\mathsf T}(\mathbf K\mathbf h_w)\lambda'_w\mathbf N_p
-\mathbf N_p^{\mathsf T}\lambda'_w(\mathbf K\mathbf h_w)^{\mathsf T}\mathbf B_p\Big]\dd\Omega,
\label{eq:assimww}
\end{equation}
\begin{equation}
\begin{aligned}
\mathbf J_{aa}-\mathbf J_{aa}^{\mathsf T}={}&-\Delta t\int\Big[\mathbf B_p^{\mathsf T}(\mathbf K\mathbf h_a)\lambda'_a\mathbf N_p
-\mathbf N_p^{\mathsf T}\lambda'_a(\mathbf K\mathbf h_a)^{\mathsf T}\mathbf B_p\Big]\dd\Omega\\
&-\Delta t\int c_a\lambda_a\Big[\mathbf N_p^{\mathsf T}\big(\mathbf K(\mathbf h_a+\nabla p_a)\big)^{\mathsf T}\mathbf B_p
-\mathbf B_p^{\mathsf T}\mathbf K(\mathbf h_a+\nabla p_a)\,\mathbf N_p\Big]\dd\Omega .
\end{aligned}
\label{eq:assimaa}
\end{equation}
$\lambda'_\pi=0$ no trecho saturado, mas o segundo termo de \eqref{eq:assimaa} existe onde
$\mathbf h_a+\nabla p_a\neq\mathbf0$, o que inclui $p_a$ uniforme com $\gamma_a>0$.

\textbf{(5) Bloco água--ar.} Pelas Eqs.~\eqref{eq:Jwa} e \eqref{eq:Jaw}, os termos $\pm\Gamma\Theta$ e os condensados se
cancelam e sobra
\begin{equation}
\begin{aligned}
\mathbf J_{wa}-\mathbf J_{aw}^{\mathsf T}={}&\int\mathbf N_p^{\mathsf T}\Big[\eta\,\Gamma p_c-\frac{n}{K_w}\Gamma\Delta p_w
+\Delta t\,c_a\lambda'_a\,\mathrm{adv}\Big]\mathbf N_p\,\dd\Omega\\
&-\Delta t\int\Big[\mathbf B_p^{\mathsf T}(\mathbf K\mathbf h_w)\lambda'_w\mathbf N_p
+\mathbf N_p^{\mathsf T}\lambda'_a(\mathbf K\mathbf h_a)^{\mathsf T}\mathbf B_p\Big]\dd\Omega,
\end{aligned}
\label{eq:assimwa}
\end{equation}
em que se usou $S_w\chi_a-S_a\chi_w=\Gamma p_c$.

\textbf{Consequência.} $\mathbf J$ não é simétrico em nenhum problema com ar incógnito, nem no limite saturado sem trocar o
sinal das linhas de massa. O código resolve o sistema linear com LU esparsa não simétrica
\loc{Projects/UnsatCommon/UnsatAnalysis.h}{34, 40}. Nos cálculos de fator de segurança, com pressões congeladas, só
$\mathbf J_{uu}$ entra no sistema filtrado e ele é simétrico; o projeto mecânico apagado (\path{SlopeMohrCoulomb}) usava
por isso uma fatoração $\mathbf L\mathbf D\mathbf L^{\mathsf T}$ (README do projeto, branch
\code{claude/relaxed-volta-1r4gog}). Os testes deste branch usam a LU em todos os casos.

\section{Forma matricial e solução não linear}\label{sec:newton}
%======================================================================

Em cada passo $t_n\to t_{n+1}$ resolve-se $\mathbf R(\mathbf X^{n+1})=\mathbf 0$ por Newton:
\begin{equation}
\mathbf R=\begin{Bmatrix}\mathbf R_u\\\mathbf R_w\\\mathbf R_a\end{Bmatrix},\qquad
\mathbf X=\begin{Bmatrix}\mathbf U\\\mathbf P_w\\\mathbf P_a\end{Bmatrix},\qquad
\mathbf J(\mathbf X^k)\,\delta\mathbf X=-\mathbf R(\mathbf X^k),\qquad \mathbf X^{k+1}=\mathbf X^k+s\,\delta\mathbf X,
\label{eq:newton}
\end{equation}
com $\mathbf J$ de \eqref{eq:Jblocos} e $s$ o passo da busca linear. É o esquema das notas \nota{§13.7} com as linhas de
massa multiplicadas por $\Delta t$. Sem plasticidade e com os coeficientes constantes no passo, $\mathbf J$ é a matriz
$\mathbf A+\theta\Delta t\,\mathbf H$ do método $\theta$ das notas com $\theta=1$ \nota{§13.6}:
\begin{equation}
\mathbf J=\begin{bmatrix}\mathbf K&-\mathbf Q_{uw}&-\mathbf Q_{ua}\\
\mathbf Q_{wu}&\mathbf C_{ww}+\Delta t\,\mathbf H_w&\mathbf C_{wa}\\
\mathbf Q_{au}&\mathbf C_{aw}&\mathbf C_{aa}+\Delta t\,\mathbf H_a\end{bmatrix},\qquad
\mathbf H_\pi=\int\mathbf B_p^{\mathsf T}\lambda_\pi\mathbf K\mathbf B_p\,\dd\Omega .
\label{eq:Jlin}
\end{equation}
Detalhes do código \loc{Projects/UnsatCommon/UnsatAnalysis.h}{135--213}:
\begin{itemize}
\item \textbf{Condições de Dirichlet.} Os valores prescritos são escritos em $\mathbf X$ antes das iterações e as equações
correspondentes são eliminadas por um filtro (\code{SetFilter}); $\delta\mathbf X=0$ nelas. Linhas 142--144 e 295--313.
\item \textbf{Convergência.} Termina quando, a partir da segunda iteração,
\begin{equation}
e_u=\frac{\|\mathbf R_u\|}{\|\mathbf F_g\|}<10^{-7}\qquad\text{e}\qquad \max_i|\delta p_i|<10^{-6}\ \text{kPa},
\label{eq:conv}
\end{equation}
em que $\|\mathbf F_g\|$ é a norma do peso nas equações de deslocamento, com um mínimo dado por
\code{SetForceReference} (usado nos testes sem peso). Linhas 145--156, 180 e 359. A norma do resíduo de massa não entra no critério \disc{D19}.
\item \textbf{Busca linear.} Nos passos transientes em que $\max|\delta p|$ ainda supera a tolerância, $s$ é dividido
por 2, até quatro vezes, enquanto a norma do resíduo das pressões $\|\mathbf R_p\|$ não diminui. Linhas 189--202.
\item \textbf{Limites.} No máximo 40 iterações (25 com pressões congeladas); o passo falha se $e_u>10^6$ ou se aparece um
valor não finito. Linhas 179 e 360.
\item \textbf{Sistema linear.} Matriz esparsa e LU (\code{TPZSpStructMatrix}, \code{ELU}). Linhas 34 e 40.
\item \textbf{Aceitação.} A memória (estado plástico, $p^n$, $S_w^n$, pressões dos vértices) só é atualizada depois da
convergência (\code{Accept}). Linhas 208--213.
\end{itemize}

\section{Fator de segurança: SRM e GIM}\label{sec:fs}
%======================================================================

Os dois métodos usam as pressões de um instante congeladas: só o equilíbrio é resolvido, com $\mathbf J_{uu}$
\loc{Projects/UnsatCommon/UnsatFS.h}{1--122}. O fator de segurança é o maior parâmetro com que o Newton ainda converge.

\textbf{Busca do colapso.} O parâmetro percorre uma lista crescente; no primeiro valor sem convergência, faz-se bisseção
entre o último valor convergido e o que falhou, até a tolerância (0,002 nos testes). O valor que falha é tentado de novo a
partir do último estado convergido. \loc{UnsatFS.h}{13--40}

\textbf{SRM.} A partir do estado sem tensões, o peso e $p_B$ são aplicados juntos em quatro incrementos com a resistência
aumentada, $F_0=0{,}6$ (estado geostático). Depois $F$ cresce na lista $0{,}8;\,1{,}0;\,\dots;\,4{,}0$, com
\begin{equation}
c_F=\frac{c}{F},\qquad \tan\phi_F=\frac{\tan\phi}{F},\qquad \tan\psi_F=\frac{\tan\psi}{F},\qquad
\mathrm{FS}_{\mathrm{SRM}}=\max\{F:\ \text{Newton converge}\}.
\label{eq:srm}
\end{equation}
\loc{UnsatFS.h}{59--98}

\textbf{GIM.} A partir do estado sem tensões, com a resistência real, o fator $\lambda$ multiplica o peso e a parte
positiva de $p_B$; a sucção ($p_B\le0$) mantém seu valor:
\begin{equation}
\lambda_g=\lambda,\qquad \lambda_p=\begin{cases}\lambda,&p_B>0\\1,&p_B\le0\end{cases},\qquad
\mathrm{FS}_{\mathrm{GIM}}=\max\{\lambda:\ \text{Newton converge}\}.
\label{eq:gim}
\end{equation}
\loc{UnsatFS.h}{100--122} \loc{TPZMatUnsaturatedUPwPa.cpp}{233}

\textbf{Relação entre os dois fatores no solo seco.} O critério de Mohr--Coulomb é homogêneo de grau 1 em $(\sig,c)$:
$\Phi(\sig/\lambda;\,c/\lambda,\phi)=\Phi(\sig;\,c,\phi)/\lambda$, com $\Phi=(\sigma_1-\sigma_3)+(\sigma_1+\sigma_3)\sn-2c\cs$.
Se $\sig$ equilibra o peso $\lambda\gamma$ com resistência $(c,\phi)$, então $\sig/\lambda$ equilibra $\gamma$ com
resistência $(c/\lambda,\phi)$. Aumentar o peso equivale, portanto, a reduzir só a coesão:
\begin{equation}
\mathrm{FS}_{\mathrm{GIM}}=\max\Big\{\lambda:\ \text{o talude com }\gamma\text{ e }\Big(\frac{c}{\lambda},\phi\Big)\text{ é estável}\Big\}
\ \ge\ \mathrm{FS}_{\mathrm{SRM}}\quad(\phi>0),
\label{eq:gimsrm}
\end{equation}
como já registrava o README do projeto mecânico apagado. Com pressões, a parte positiva de $p_B$ escalada preserva essa
equivalência; a sucção não escalada a quebra, por escolha.

\textbf{Talude seco de referência.} Altura 10~m, inclinação 45°, $c=10$~kPa, $\phi=30^\circ$, $\gamma=20$~kN/m$^3$,
$E=2\times10^4$~kPa, $\nu=0{,}3$; malha estruturada com $n_f$ colunas na zona da face (\code{SlopeMeshFace}). As
referências são o Bishop simplificado e o modelo puramente mecânico com refino adaptativo, ambos do README do projeto
\path{SlopeMohrCoulomb} (branch \code{claude/relaxed-volta-1r4gog}).
\begin{center}\small

malha & equações & FS$_{\mathrm{SRM}}$ & FS$_{\mathrm{GIM}}$\\

\textbf{Observação sobre a referência de Gardner.} Li et al.\ (2024, Eq.~9, p.~6) escrevem
$(u_a-u_w)=-\frac1\alpha\ln\big[(1+\frac Q{k_s})e^{-\gamma_w\alpha z}-\frac Q{k_s}\big]$, com $z$ a altura acima do nível
d'água, e afirmam: ``For infiltration, $Q>0$, whereas for evaporation, $Q<0$''. Com $Q>0$ o argumento do logaritmo é
menor que $e^{-\gamma_w\alpha z}$ e a sucção fica acima da hidrostática $\gamma_wz$, o que corresponde a fluxo para cima; com
os dados do teste 2.2 ($\alpha=0{,}2$~kPa$^{-1}$, $Q=0{,}3k_s$, $z=3$~m) o argumento é negativo. A Eq.~\eqref{eq:gardner},
que a solução de elementos finitos reproduz com a vazão entrando pelo topo, é a Eq.~9 de Li et al.\ com $Q=-q$.

\textbf{Observação sobre a referência de Mandel.} No apêndice A.1 de Haghighat et al.\ (2021, arXiv:2110.03049v1, p.~25),
o coeficiente da solução é $c_f=k/[\mu(1/M+b^2/K_{dr})]$, e $K_{dr}$ é definido na p.~5 como o módulo volumétrico drenado.
Com os parâmetros do teste 4b ($E=10^4$~kPa, $\nu=0{,}3$, $n=0{,}3$, $K_w=2{,}2\times10^6$~kPa, $k_s=10^{-6}$~m/s), esse
coeficiente vale $8{,}49\times10^{-4}$~m$^2$/s; com o módulo confinado $\lambda+2G$ no lugar de $K_{dr}$, vale
$1{,}37\times10^{-3}$~m$^2$/s. A solução de elementos finitos concorda com a solução analítica avaliada com o segundo
valor (erro no centro $\le0{,}004\,p_0$); com o primeiro, o erro no centro chega a $0{,}23\,p_0$ em $c_vt/a^2=0{,}6$ (saída
do teste 4b). O teste usa o módulo confinado:
\begin{equation}
c_v=\frac{k_s}{\gamma_w}\,\frac{M(\lambda+2G)}{\lambda+2G+\alpha^2M},\qquad \frac1M=\frac{n}{K_w}\ (\text{grãos incompressíveis}).
\label{eq:cmandel}
\end{equation}
