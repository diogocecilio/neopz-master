kernel[x1_, x2_] := Exp[-Abs[x1[[1]] - x2[[1]]]/LX - Abs[x1[[2]] - x2[[2]]]/LY];
mesh = ToElementMesh[FullRegion[2], {{0, 5}, {0, 1}}, "MeshOrder" -> 1, MaxCellMeasure -> 0.0025];

CreateField[LX_, LY_, M_] := Block[{CMAT, els, w, sqrtW, KN, lambda, eigenvectors},
  CMAT = Table[kernel[coords[[i]], ct], {i, 1, Nq}];
  els = mesh["MeshElements"][[1, 1]];
  w = ConstantArray[0., Nq];
  Do[w[[t]] += Area[Polygon[coords[[t]]]]/Length[t], {t, els}];
  sqrtW = DiagonalMatrix[Sqrt[w]];
  KN = sqrtW . CMAT . sqrtW;
  {lambda, eigenvectors} = Eigensystem[KN, M];
  {w, lambda, eigenvectors}]

(* =====6. O MEF=====*)
DmatEPT[EE_, nu_] := EE/(1 - nu^2) {{1, nu, 0}, {nu, 1, 0}, {0, 0, (1 - nu)/2}}   (*estado plano de tensao:sigma_zz=0*)
Dmat[EE_, nu_] := EE/((1 + nu) (1 - 2 nu)) {{1 - nu, nu, 0}, {nu, 1 - nu, 0}, {0, 0, (1 - 2 nu)/2}}   (*estado plano de deformacao:eps_zz=0*)
voigt[i_, k_] := If[i == k, i, 3]
cMat[Dm_, i_, j_] := Table[Dm[[voigt[i, k], voigt[j, l]]], {k, 1, 2}, {l, 1, 2}]
opElast[Dm_] := Table[Sum[Inactive[Div][(-cMat[Dm, i, j]) . Inactive[Grad][{u[x, y], v[x, y]}[[j]], {x, y}], {x, y}], {j, 1, 2}], {i, 1, 2}]
apoioV = DirichletCondition[v[x, y] == 0., x == 0 || x == 5];   (*v=0 nas faces das pontas:o apoio simples*)
travaU = DirichletCondition[u[x, y] == 0., x == 0 && y == 0];   (*u=0 so no canto (0,0):tira a translacao horizontal*)
meshFE = ToElementMesh[FullRegion[2], {{0, 5}, {0, 1}}, "MeshOrder" -> 2,(*a malha do MEF:ordem 2,senao a flexao fica rigida demais*)
   MaxCellMeasure -> 0.05];

(* =====Solucoes analiticas:viga biapoiada,carga q (para baixo) na face de cima=====*)
(*x em[0,L],y em[0,h],espessura 1;devolvem v (negativo=para baixo),como o vfun*)
(*1) Euler-Bernoulli:so flexao*)
fEB[x_, q_, L_, h_, EE_] := -q x (L^3 - 2 L x^2 + x^3)/(24 EE h^3/12)
(*2) viga de Timoshenko:flexao+cortante,kappa=5/6*)
fTim[x_, q_, L_, h_, EE_, nu_] := fEB[x, q, L, h, EE] - q x (L - x)/(2 (5/6) EE/(2 (1 + nu)) h)
(*3) Timoshenko& Goodier:elasticidade 2D exata em estado plano de tensao,v(x,y) em todo o corpo*)
fanal[x_, y_, q_, L_, h_, EE_, nu_] := -q/(160 EE h^3) (h^4 (-15 + 2 nu) + 32 h^3 nu y +
    192 h^2 (L x - x^2 - nu y^2) + 160 h y (3 L nu x - 3 nu x^2 + y^2 + 2 nu y^2) +
    80 (L^3 x + x^4 + 6 nu x^2 y^2 - (1 + 2 nu) y^4 - 2 L (x^3 + 3 nu x y^2)))
(*o deslocamento horizontal u(x,y) de Timoshenko& Goodier,para comparar com o ufun*)
uanal[x_, y_, q_, L_, h_, EE_, nu_] := -q (L - 2 x)/(20 EE h^3) (2 h^3 - 24 h^2 y -
    10 y (L^2 + 2 L x - 2 x^2 + 2 (2 + nu) y^2) + 5 h (L^2 + 2 L x - 2 x^2 + 6 (2 + nu) y^2))
{ufun, vfun} = NDSolveValue[{opElast[DmatEPT[200000, 0.3]] == {0, NeumannValue[-1, 0 <= x <= 5 && y == 1]}, apoioV, travaU}, {u, v}, {x, y} \[Element] meshFE];
Print["meio do vao (2.5, 0.5): MEF = ", vfun[2.5, 0.5], "  EB = ", fEB[2.5, 1, 5, 1, 200000],
  "  Timoshenko = ", fTim[2.5, 1, 5, 1, 200000, 0.3], "  T&G 2D = ", fanal[2.5, 0.5, 1, 5, 1, 200000, 0.3]];
Print["u em (0, 1): MEF = ", ufun[0, 1], "  T&G 2D = ", uanal[0, 1, 1, 5, 1, 200000, 0.3]];
Plot[{vfun[x, 0.5], fEB[x, 1, 5, 1, 200000], fanal[x, 0.5, 1, 5, 1, 200000, 0.3]}, {x, 0, 5},
 PlotLegends -> {"MEF", "Euler-Bernoulli", "T&G 2D"}, AxesLabel -> {"x", "v(x, 0.5)"},

hDif = 0.1;                                    (*o passo das diferencas centrais em xi*)
nAnalises = 0;                                 (*contador de analises do MEF*)
gXi[xi_] := (nAnalises += 1; deltaLim - flechaDe[xi])   (*g(xi)=deltaLim-delta:falha se g<=0*)
gradXi[xi_] := Module[{gs}, nAnalises += 2 M;
  gs = ParallelMap[flechaDe, Join[Table[xi + hDif UnitVector[M, k], {k, M}], Table[xi - hDif UnitVector[M, k], {k, M}]]];   (*2M analises do MEF,em paralelo*)
  (gs[[M + 1 ;;]] - gs[[;; M]])/(2 hDif)]      (*grad g=-grad delta,Eq.(difc)*)
formXi[gf_, gradf_, n_, maxiter_ : 20, tol_ : 10^-6] := Module[{xi = ConstantArray[0., n], g0, g, grad, xiNovo, beta = 0.},
  g0 = g = gf[xi];(*comeca na origem,xi=0*)
  Do[
   grad = gradf[xi];(*o gradiente:uma vez por iteracao*)
   xiNovo = (grad . xi - g)/(grad . grad) grad;(*Eq.(hlrf):o ponto do plano tangente mais perto da origem*)
   g = gf[xiNovo];
   Print[" | iter = ", it, " | g/g0 = ", g/g0, " | beta = ", Norm[xiNovo]];
   If[Abs[g/g0] < tol && Abs[Norm[xiNovo] - beta] < 0.001, xi = xiNovo; Break[]];(*para:g~0 e beta parado*)
   xi = xiNovo; beta = Norm[xi], {it, maxiter}];
  grad = gradf[xi];(*o gradiente no ponto de projeto,para o alpha*)
  <|"xi" -> xi,(*o ponto de projeto xi**)"beta" -> Sign[g0] Norm[xi],(*beta<0 se a origem ja falha*)
   "alpha" -> grad/Norm[grad],(*os cossenos diretores,Eq.(ponto)*)"g" -> g|>]

(* =====13. SORM: a curvatura da superficie de falha no ponto de projeto=====*)
(*Perto do ponto de projeto xiStar, g(xi) ~ grad.(xi - xiStar) + (1/2)(xi - xiStar).H.(xi - xiStar). Com a normal
  n = -grad/|grad| (da origem para a falha, xiStar = beta n) e uma base R do plano tangente, as curvaturas principais kappa_i sao os autovalores de
  R.H.R^T/|grad| (M - 1 curvaturas). Com elas:
    Breitung (1984):               P_f = Phi(-beta) Prod (1 + beta kappa_i)^(-1/2)
    Hohenbichler-Rackwitz (1988):  P_f = Phi(-beta) Prod (1 + psi kappa_i)^(-1/2),  psi = phi(beta)/Phi(-beta)
  kappa < 0: a regiao de falha e maior que o semiespaco do FORM, e a P_f do SORM fica acima da do FORM.
  O hessiano de g = deltaLim - delta sai por diferencas finitas no ponto de projeto, com passo hS: 2M analises nos eixos
  (o diagonal e o gradiente) e 2 por par (i, j), xi* + hS(e_i + e_j) e xi* - hS(e_i + e_j): M(M - 1) + 2M + 1 analises*)
hS = 0.2;                                              (*o passo do hessiano em xi*)
hessianoXi[xi_] := Module[{n = Length[xi], d0, dp, dm, ij, dpp = {}, dmm = {}, Hd},
  d0 = flechaDe[xi];
  {dp, dm} = Partition[ParallelMap[flechaDe, Join[Table[xi + hS UnitVector[n, k], {k, n}], Table[xi - hS UnitVector[n, k], {k, n}]]], n];
  ij = Subsets[Range[n], {2}];                         (*os pares i < j*)
  If[ij =!= {}, {dpp, dmm} = Partition[ParallelMap[flechaDe,
      Join[(xi + hS (UnitVector[n, #[[1]]] + UnitVector[n, #[[2]]])) & /@ ij, (xi - hS (UnitVector[n, #[[1]]] + UnitVector[n, #[[2]]])) & /@ ij]],
      Length[ij]]];
  Hd = DiagonalMatrix[(dp - 2 d0 + dm)/hS^2];          (*d2 delta/dxi_i^2*)
  Do[With[{i = ij[[k, 1]], j = ij[[k, 2]]},           (*d2 delta/dxi_i dxi_j, com erro O(hS^2)*)
    Hd[[i, j]] = Hd[[j, i]] = (dpp[[k]] + dmm[[k]] - dp[[i]] - dm[[i]] - dp[[j]] - dm[[j]] + 2 d0)/(2 hS^2)], {k, Length[ij]}];
  <|"Hg" -> -Hd, "grad" -> -(dp - dm)/(2 hS), "analises" -> 1 + 2 n + 2 Length[ij]|>]   (*o hessiano e o gradiente de g*)
rodaSORM[m_] := Module[{arq = FileNameJoin[{pasta, "sorm_M" <> ToString[m] <> "_d" <> ToString[deltaLim, InputForm] <> "_h" <> ToString[hS] <> ".mx"}],
   hs, rF = formsM[m], gr, nrm, nh, R, kap, beta, psi, phiB},
  If[FileExistsQ[arq], hs = Import[arq], usaModos[m]; hs = hessianoXi[rF["xi"]]; Export[arq, hs]];   (*o hessiano so e calculado uma vez*)
  beta = rF["beta"]; gr = hs["grad"]; nrm = Norm[gr];
  nh = -gr/nrm;                                        (*a normal, da origem para a falha*)
  R = Orthogonalize[NullSpace[{nh}]];                  (*base ortonormal do plano tangente: (M - 1) x M*)
  kap = Eigenvalues[(# + Transpose[#])/2 &[R . hs["Hg"] . Transpose[R]/nrm]];   (*as M - 1 curvaturas principais*)
  phiB = CDF[NormalDistribution[], -beta]; psi = PDF[NormalDistribution[], beta]/phiB;
  <|"kappa" -> kap, "analises" -> hs["analises"],
   "PfB" -> If[Min[1 + beta kap] > 0, phiB Times @@ ((1 + beta kap)^(-1/2)), Indeterminate],   (*Breitung*)
   "PfHR" -> If[Min[1 + psi kap] > 0, phiB Times @@ ((1 + psi kap)^(-1/2)), Indeterminate],   (*Hohenbichler-Rackwitz*)
   "angulo" -> ArcCos[Clip[nh . rF["xi"]/Norm[rF["xi"]], {-1, 1}]]/Degree|>]   (*confere: a normal aponta para o xi*?*)
sormM = AssociationMap[rodaSORM, listaMF];             (*cuidado: com M = 200 sao ~40 000 analises (uns 20 a 30 min)*)
usaModos[Mcampo];                                      (*volta ao campo de 200*)

(* =====14. AI com o SORM (M = 20): a normal centrada no xi*, com a covariancia das curvaturas=====*)
(*Perto do ponto de projeto, a falha e u_n >= beta + (1/2) y.K.y (u_n na normal, y no plano tangente, K as curvaturas).
  Integrando u_n, as falhas se distribuem em y como exp(-(1/2) y.(I + c K).y), com c = beta (Breitung) ou c = psi
  (Hohenbichler-Rackwitz): uma normal com covariancia (I + c K)^-1 no plano tangente. A AI com o SORM sorteia
  xi ~ N(xi*, Sigma), com Sigma = n n^T + R^T (I + c K)^-1 R nos mS primeiros modos (os do FORM e do SORM) e a identidade
  nos outros. Como as curvaturas sao negativas, ela abre a amostra nas direcoes em que a falha se espalha.
  O peso e o exato da normal: ln w = -|xi|^2/2 + |z|^2/2 + ln det(Sigma)/2, com xi = xi* + z.U e Sigma = U^T U*)
mS = 20;                                               (*o M do FORM e do SORM*)
sigmaSORM[m_, tipo_] := Module[{hs = Import[FileNameJoin[{pasta, "sorm_M" <> ToString[m] <> "_d" <> ToString[deltaLim, InputForm] <> "_h" <> ToString[hS] <> ".mx"}]],
   beta = formsM[m]["beta"], gr, nrm, nh, R, At, kap, vec, c},
  gr = hs["grad"]; nrm = Norm[gr]; nh = -gr/nrm;       (*a normal, da origem para a falha*)
  R = Orthogonalize[NullSpace[{nh}]];                  (*base do plano tangente: (m - 1) x m*)
  At = (# + Transpose[#])/2 &[R . hs["Hg"] . Transpose[R]/nrm];
  {kap, vec} = Eigensystem[At];                        (*as curvaturas e as direcoes principais*)
  c = If[tipo == "HR", PDF[NormalDistribution[], beta]/CDF[NormalDistribution[], -beta], beta];
  (# + Transpose[#])/2 &[Outer[Times, nh, nh] + Transpose[R] . Transpose[vec] . DiagonalMatrix[1/(1 + c kap)] . vec . R]]   (*Sigma, m x m*)
rodaAISORM[tipo_] := Module[{arq = FileNameJoin[{pasta, "ai_M200_sorm" <> ToString[mS] <> tipo <> "_N1000_d" <> ToString[deltaLim, InputForm] <> "_s2.mx"}],
   mu = PadRight[formsM[mS]["xi"], Mcampo], S, U, Z, xs, res, t, ds, iw},
  S = sigmaSORM[mS, tipo];
  U = ArrayFlatten[{{CholeskyDecomposition[S], 0}, {0, IdentityMatrix[Mcampo - mS]}}];   (*Sigma de 200 = U^T U*)
  If[FileExistsQ[arq], res = Import[arq],
   SeedRandom[2];
   Z = RandomVariate[NormalDistribution[0, 1], {1000, Mcampo}];   (*os mesmos z da AI pela media (numeros aleatorios comuns)*)
   xs = (mu + #) & /@ (Z . U);                         (*xi ~ N(xi*, Sigma)*)
   {t, ds} = AbsoluteTiming[ParallelMap[flechaDe, xs]];
   res = <|"flechas" -> ds, "logpeso" -> -Total[xs^2, {2}]/2 + Total[Z^2, {2}]/2 + Log[Det[S]]/2, "tempo" -> t|>;   (*ln w exato*)
   Export[arq, res]];
  iw = (Boole[# > deltaLim] & /@ res["flechas"]) Exp[res["logpeso"]];
  <|"Pf" -> Mean[iw], "CoV" -> StandardDeviation[iw]/(Sqrt[1000] Mean[iw]), "falhas" -> Count[res["flechas"], d_ /; d > deltaLim],
   "ESS" -> Total[iw]^2/Total[iw^2], "maior" -> Max[iw]/Total[iw], "S" -> S|>]
(*o CoV verdadeiro de cada densidade, pelas falhas do MC cru de 50000 (sem MEF): Var_h(I w) = P_f E[w | falha] - P_f^2*)
nAI5gauss[mu20_, S_] := With[{x20 = xiFalhas[[All, ;; mS]], Si = Inverse[S]},
  With[{lw = -Total[x20^2, {2}]/2 + Total[((x20 - ConstantArray[mu20, Length[x20]]) . Si) (x20 - ConstantArray[mu20, Length[x20]]), {2}]/2 + Log[Det[S]]/2},
   (pf50 Mean[Exp[lw]] - pf50^2)/(0.05 pf50)^2]]
usaModos[Mcampo];                                      (*a AI usa o campo de 200*)
aiSORM = AssociationMap[rodaAISORM, {"B", "HR"}];      (*Breitung (c = beta) e Hohenbichler-Rackwitz (c = psi)*)
nHess = sormM[mS]["analises"]; nF = formsM[mS]["analises"];
aiMedia = With[{a = Import[FileNameJoin[{pasta, "ai_M200_f" <> ToString[mS] <> "_N1000_d" <> ToString[deltaLim, InputForm] <> "_s2.mx"}]]},
  With[{iw = (Boole[# > deltaLim] & /@ a["flechas"]) a["pesos"]},   (*a AI pela media no xi* do FORM de 20, a da secao 12*)
   <|"Pf" -> Mean[iw], "CoV" -> StandardDeviation[iw]/(Sqrt[1000] Mean[iw]), "falhas" -> Count[a["flechas"], d_ /; d > deltaLim],
    "ESS" -> Total[iw]^2/Total[iw^2], "maior" -> Max[iw]/Total[iw]|>]];

Grid[{{"AI com o campo de 200 (1000 amostras)", "P_f", "CoV (%)", "erro (%)", "z", "falhas", "ESS_F", "maior parcela",

(*FORM: xi ~ N(xi*, I). SORM: xi ~ N(xi*, Sigma), com o Sigma das curvaturas (secao 14), Breitung ou H-R. A AI usa sempre o
  campo de 200, e as duas densidades mudam so os mS primeiros modos. Tres verificacoes:
  a) o CoV verdadeiro de cada densidade, para cada M, pelas falhas do MC cru de 50000 (sem MEF);
  b) 10 sementes independentes de 1000 amostras de cada densidade com M = mS, com as mesmas sementes nas tres (numeros
     aleatorios comuns): a media, a dispersao entre as sementes, o CoV estimado e a diferenca SORM - FORM, semente a semente;
  c) a convergencia da P_f e do CoV, com as 10 sementes em sequencia*)
convPf[iw_] := Accumulate[iw]/N[Range[Length[iw]]]    (*a P_f acumulada*)
convCoV[iw_] := Module[{k = Range[Length[iw]], pf = convPf[iw], vr},
  vr = (Accumulate[iw^2] - k pf^2)/(k - 1 /. 0 -> 1);   (*a variancia amostral ate a amostra k*)
  MapThread[If[#1 > 0 && #3 > 1, Sqrt[Max[#2, 0]/#3]/#1, Indeterminate] &, {pf, vr, k}]]

(* =====16. Line sampling: linhas paralelas a direcao do FORM, com o campo de 200=====*)
(*A direcao e = xi*/beta do FORM de m termos (completada com zeros ate 200). Cada amostra e uma linha: sorteia
  z ~ N(0, I) com 200 termos, tira a componente em e (zp = z - (z.e) e) e acha, na reta zp + c e, o cStar em que a flecha
  vale deltaLim. Como zp e a coordenada c sao independentes, P_f = E[Phi(-cStar)], e cada linha da p_i = Phi(-cStar).
  So a variacao de cStar de uma linha para outra (a nao linearidade da superficie de falha) entra na variancia.
  A raiz e pela secante em g(c) = ln(flecha/deltaLim), quase linear em c (a flecha ~ 1/E, E lognormal): a 1a analise em
  c = beta, a 2a no passo de Newton com a inclinacao s medida no ponto de projeto (2 analises por direcao), e a raiz
  da secante e aceita sem nova analise quando o passo for menor que tolLS. O estimador nao tem vies para nenhuma
  direcao: uma direcao ruim so aumenta a dispersao das p_i*)
nLS = 500;                                             (*linhas por direcao*)
tolLS = 0.01;                                          (*passo da secante aceito sem nova analise (em c)*)
listaLS = {5, 8, 20, 200};                             (*o M do FORM que da a direcao*)
gLinha[xi_] := Log[flechaDe[xi]/deltaLim]              (*> 0: falha*)
raizLinha[zp_, e_, c0_, s_] := Module[{ca = c0, ga, cb, gb, cn, n = 2},
  ga = gLinha[zp + ca e];
  cb = Clip[ca - ga/s, {-8., 12.}]; gb = gLinha[zp + cb e];   (*o passo de Newton com a inclinacao do ponto de projeto*)
  cn = Clip[cb - gb (cb - ca)/(gb - ga), {-8., 12.}];
  While[Abs[cn - cb] > tolLS && n < 8,                 (*secante ate o passo ficar menor que tolLS*)
   ca = cb; ga = gb; cb = cn; gb = gLinha[zp + cb e]; n++;
   cn = Clip[cb - gb (cb - ca)/(gb - ga), {-8., 12.}]];
  {cn, n}]                                             (*{cStar, analises do MEF da linha}*)
usaModos[Mcampo];
DistributeDefinitions[gLinha, raizLinha, tolLS];
rodaLS[m_] := Module[{arq = FileNameJoin[{pasta, "ls_M200_f" <> ToString[m] <> "_N" <> ToString[nLS] <> "_d" <> ToString[deltaLim, InputForm] <> "_s2.mx"}],
   mu = PadRight[formsM[m]["xi"], Mcampo], beta, e, s, Z, zp, t, rs, res, p},
  beta = Norm[mu]; e = mu/beta;
  If[FileExistsQ[arq], res = Import[arq],
   s = (gLinha[mu + 0.1 e] - gLinha[mu - 0.1 e])/0.2;  (*a inclinacao de g ao longo de e, no ponto de projeto*)
   SeedRandom[2];
   Z = RandomVariate[NormalDistribution[0, 1], {nLS, Mcampo}];   (*os mesmos z em todas as direcoes*)
   zp = (# - (# . e) e) & /@ Z;                        (*a parte de z ortogonal a e*)
   {t, rs} = AbsoluteTiming[ParallelMap[With[{ee = e, c0 = beta, ss = s}, raizLinha[#, ee, c0, ss] &], zp]];
   res = <|"c" -> rs[[All, 1]], "nLinha" -> rs[[All, 2]], "s" -> s, "beta" -> beta, "tempo" -> t|>;
   Export[arq, res]];
  p = CDF[NormalDistribution[], -res["c"]];            (*p_i de cada linha*)
  <|"p" -> p, "Pf" -> Mean[p], "CoV" -> StandardDeviation[p]/(Sqrt[nLS] Mean[p]), "nLinha" -> res["nLinha"],
   "porLinha" -> N[Mean[res["nLinha"]]], "fixas" -> formsM[m]["analises"] + 2, "s" -> res["s"], "tempo" -> res["tempo"]|>]
ls = AssociationMap[rodaLS, listaLS];                  (*~ 4 x 200 x 2,5 analises do MEF*)
linhasLS5[m_] := nLS (ls[m]["CoV"]/0.05)^2             (*linhas para CoV de 5%*)
totalLS5[m_] := ls[m]["fixas"] + ls[m]["porLinha"] linhasLS5[m]
custoAI[m_] := SelectFirst[custo, First[#] == m &][[4]]   (*o total da AI pela media (secao 12)*)
Grid[Prepend[Table[With[{a = ls[m]},
     {m, a["fixas"], NumberForm[a["Pf"], 4], NumberForm[100 a["CoV"], 3], erro[a["Pf"]], NumberForm[zDe[a["Pf"], a["CoV"]], 2],
      NumberForm[a["porLinha"], 3], Round[linhasLS5[m]], Round[totalLS5[m]], NumberForm[nMC5/totalLS5[m], 3], Round[custoAI[m]]}], {m, listaLS}],
   {"direcao: FORM de M", "analises fixas (FORM + 2)", "P_f LS (" <> ToString[nLS] <> " linhas)", "CoV (%)", "erro vs cru (%)", "z",
    "analises por linha", "linhas p/ CoV 5%", "total p/ CoV 5%", "ganho vs MC cru", "total AI (secao 12)"}],
  Frame -> All, Background -> {None, {1 -> LightGray}}]
Print["MC cru de 50000: P_f = ", pf50, " (CoV ", cv50, ", ", Round[nMC5], " analises p/ CoV 5%); P_f FORM: ",
  Table[m -> pfFORMde[m], {m, listaLS}]];
(*a convergencia contra as analises do MEF (contando o FORM): line sampling x AI pela media no mesmo xi* do FORM*)
convLS[m_] := With[{p = ls[m]["p"], k = Range[nLS], x = ls[m]["fixas"] + Accumulate[ls[m]["nLinha"]]},
  With[{pm = Accumulate[p]/k, vr = (Accumulate[p^2] - k (Accumulate[p]/k)^2)/(k - 1 /. 0 -> 1)},
   {Transpose[{x, pm}], Transpose[{x, Sqrt[Clip[vr, {0, Infinity}]/k]/pm}][[3 ;;]]}]]
convAI[m_] := Quiet@With[{a = Import[FileNameJoin[{pasta, "ai_M200_f" <> ToString[m] <> "_N1000_d" <> ToString[deltaLim, InputForm] <> "_s2.mx"}]]},

    {Transpose[{formsM[m]["analises"] + k, pm}],
     Select[Transpose[{formsM[m]["analises"] + k, Sqrt[Clip[vr, {0, Infinity}]/k]/pm}], #[[2]] > 0 &]}]]]
mConv = {8, 20};
gLSPf = ListLogLinearPlot[Join[convLS[#][[1]] & /@ mConv, convAI[#][[1]] & /@ mConv], Joined -> True,

(* =====17. O gradiente adjunto: o FORM com os 200 termos por 2 analises por iteracao, e o line sampling nessa direcao=====*)
(*A flecha e delta = -v(2.5, 0.5). Com a carga adjunta (uma forca unitaria para baixo no ponto, aqui uma gaussiana
  estreita), a(w, lam) = delta(w) para todo w. Derivando a(u, w) = l(w) em xi_i e tomando w = lam:
  d delta/d xi_i = -Integral[eps(u).D'.eps(lam)], com D' = D d(ln E)/d xi_i = D sg Sqrt[lambda_i] f_i(x). Entao
  d delta/d xi_i = -sg Sqrt[lambda_i] Sum_j w_j f_i(x_j) e(x_j), e = eps(u).D(E).eps(lam) nos nos de Nystrom: os 200 termos
  com 2 analises (a direta e a adjunta), contra 401 das diferencas centrais. No ponto fixo do HL-RF, xi* e paralelo ao
  gradiente, entao um erro de escala da carga adjunta nao muda o ponto de projeto*)
sAdj = 0.15;                                           (*a largura da gaussiana da carga adjunta*)
cargaAdj[x_, y_] := Exp[-((x - 2.5)^2 + (y - 0.5)^2)/(2 sAdj^2)]/(2 Pi sAdj^2)
epsDe[uu_, vv_] := Transpose[{Derivative[1, 0][uu] @@@ coords, Derivative[0, 1][vv] @@@ coords,
    (Derivative[0, 1][uu] @@@ coords) + (Derivative[1, 0][vv] @@@ coords)}]   (*eps_xx, eps_yy, gamma_xy nos nos de Nystrom*)
gradAdj[xi_] := Module[{Ei = ElementMeshInterpolation[{meshFE}, EnosDe[xi]], uu, vv, la, lb, eu, el, En, e},
  nAnalises += 2;
  {uu, vv} = NDSolveValue[{opElast[DmatEPT[Ei[x, y], \[Nu] /. data]] == {0, NeumannValue[-1, 0 <= x <= 5 && y == 1]}, apoioV, travaU}, {u, v}, {x, y} \[Element] meshFE];
  {la, lb} = NDSolveValue[{opElast[DmatEPT[Ei[x, y], \[Nu] /. data]] == {0, -cargaAdj[x, y]}, apoioV, travaU}, {u, v}, {x, y} \[Element] meshFE];
  eu = epsDe[uu, vv]; el = epsDe[la, lb]; En = Ei @@@ coords;
  e = MapThread[#1 . DmatEPT[#3, \[Nu] /. data] . #2 &, {eu, el, En}];   (*a densidade de energia mutua*)
  sg Sqrt[lambda] (Fmat . (w e))]                      (*grad g = -grad delta = +sg Sqrt[lambda_i] Sum w f_i e*)
(*confere com as diferencas centrais na origem, com M = 8*)
usaModos[8];
With[{ga = gradAdj[ConstantArray[0., 8]], gd = gradXi[ConstantArray[0., 8]]},
 Print["gradiente na origem (M = 8): adjunto ", ga, "\n  diferencas ", gd, "\n  angulo (graus) ", VectorAngle[ga, gd]/Degree,
  ", razao das normas ", Norm[ga]/Norm[gd]]];
(*o FORM com os 200 termos*)
rodaFORMadj[m_] := Module[{arq = FileNameJoin[{pasta, "form_adj_M" <> ToString[m] <> "_d" <> ToString[deltaLim, InputForm] <> ".mx"}], res, tF, rF},
  usaModos[m];
  If[FileExistsQ[arq], res = Import[arq],
   nAnalises = 0;
   {tF, rF} = AbsoluteTiming[formXi[gXi, gradAdj, m]];
   res = <|"M" -> m, "beta" -> rF["beta"], "xi" -> rF["xi"], "alpha" -> rF["alpha"], "tempo" -> tF, "analises" -> nAnalises|>;
   Export[arq, res]];
  res]
formsM["adj200"] = rodaFORMadj[200];
usaModos[Mcampo];
